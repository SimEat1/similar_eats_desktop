/// --------------------------- HOME ---------------------------

class HomeScreen extends StatefulWidget {
  const HomeScreen({super.key});

  @override
  State<HomeScreen> createState() => _HomeScreenState();
}

class _HomeScreenState extends State<HomeScreen> {
  late final String _prompt = _prompts[Random().nextInt(_prompts.length)];
  final List<String> _wants = <String>[
    // starter examples (you’ll replace from CravingScreen)
    //'🍗 Wings',
    //'Johnny’s Wings',
  ];

  @override
  Widget build(BuildContext context) {
    final cs = Theme.of(context).colorScheme;

    return Scaffold(
      appBar: AppBar(
        title: const Text(
          'Similar Eats',
          style: TextStyle(fontWeight: FontWeight.w800, fontSize: 22),
        ),
      ),
      body: ListView(
        padding: const EdgeInsets.fromLTRB(16, 8, 16, 24),
        children: [
          // Avatar
          const SizedBox(height: 8),
          Center(
            child: CircleAvatar(
              radius: 52,
              backgroundColor: Colors.brown.shade200,
              child: const Text(
                'D',
                style: TextStyle(fontSize: 34, fontWeight: FontWeight.w800, color: Colors.brown),
              ),
            ),
          ),
          const SizedBox(height: 18),

          // Clickable prompt
          Row(
            mainAxisAlignment: MainAxisAlignment.center,
            children: [
              const Icon(Icons.restaurant_menu_rounded, color: Color(0xFFF25C54)),
              const SizedBox(width: 8),
              GestureDetector(
                onTap: () async {
                  final List<String>? picked = await Navigator.push(
                    context,
                    MaterialPageRoute(builder: (_) => const CravingScreen()),
                  );
                  if (picked != null && picked.isNotEmpty) {
                    setState(() {
                      // merge unique, keep order by adding new at end
                      for (final p in picked) {
                        if (!_wants.contains(p)) _wants.add(p);
                      }
                    });
                  }
                },
                child: Text(
                  _prompt,
                  style: const TextStyle(
                    fontSize: 20,
                    fontWeight: FontWeight.w900,
                    color: Color(0xFFF25C54),
                  ),
                ),
              ),
            ],
          ),

          const SizedBox(height: 18),
          const _SectionTitle('Dishes you want to try'),
          const SizedBox(height: 8),

          // Chips list (from selections)
          if (_wants.isEmpty)
            Opacity(
              opacity: .6,
              child: Row(
                children: [
                  _ghostChip(width: 96),
                  const SizedBox(width: 12),
                  _ghostChip(width: 160),
                ],
              ),
            )
          else
            Wrap(
              spacing: 10,
              runSpacing: 10,
              children: _wants
                  .map(
                    (w) => InputChip(
                      label: Text(w),
                      backgroundColor: Colors.white,
                      onDeleted: () {
                        setState(() => _wants.remove(w));
                      },
                    ),
                  )
                  .toList(),
            ),

          const SizedBox(height: 22),
          const _SectionTitle('Suggested matches'),
          const SizedBox(height: 8),
          for (final r in _restaurants) _RestaurantCard(r),
          const SizedBox(height: 24),
          Center(
            child: Text('Mock data • offline',
                style: TextStyle(color: cs.outline)),
          ),
        ],
      ),
    );
  }
}

    return Text(

      style: const TextStyle(fontSize: 18, fontWeight: FontWeight.w800),

  }

Widget _ghostChip({required double width}) => Container(
      height: 40,
      width: width,
      decoration: BoxDecoration(
        color: Colors.white.withOpacity(.8),
        borderRadius: BorderRadius.circular(20),
      ),

/// --------------------------- CRAVING SCREEN ---------------------------

class CravingScreen extends StatefulWidget {
  const CravingScreen({super.key});

  @override
  State<CravingScreen> createState() => _CravingScreenState();
}

class _CravingScreenState extends State<CravingScreen> {
  final List<_Cat> _categories = const [
    _Cat('🍗', 'Chicken'),
    _Cat('🍕', 'Pizza'),
    _Cat('🍔', 'Burgers'),
    _Cat('🍣', 'Sushi'),
    _Cat('🌮', 'Tacos'),
    _Cat('🥗', 'Salad'),
    _Cat('🍝', 'Pasta'),
    _Cat('🍜', 'Ramen'),
    _Cat('🍰', 'Dessert'),
    _Cat('☕', 'Coffee'),
  ];

  final Set<String> _selected = <String>{};
  final TextEditingController _custom = TextEditingController();

  void _toggle(String label) {
    setState(() {
      if (_selected.contains(label)) {
        _selected.remove(label);
      } else {
        _selected.add(label);
      }
    });
  }

  void _submit() {
    final List<String> result = _selected.toList();
    final custom = _custom.text.trim();
    if (custom.isNotEmpty) {
      result.add(custom);
    }
    Navigator.pop(context, result);
  }

  @override
  Widget build(BuildContext context) {
    final chips = _categories
        .map((c) => ChoiceChip(
              label: Text('${c.emoji} ${c.label}'),
              selected: _selected.contains('${c.emoji} ${c.label}'),
              onSelected: (_) => _toggle('${c.emoji} ${c.label}'),
            ))
        .toList();

    return Scaffold(
      appBar: AppBar(
        title: const Text(
          'What are you craving?',
          style: TextStyle(fontWeight: FontWeight.w800, fontSize: 20),
        ),
      ),
      body: Padding(
        padding: const EdgeInsets.fromLTRB(16, 12, 16, 16),
        child: Column(
          crossAxisAlignment: CrossAxisAlignment.start,
          children: [
            Wrap(spacing: 10, runSpacing: 10, children: chips),
            const SizedBox(height: 18),
            TextField(
              controller: _custom,
              textInputAction: TextInputAction.done,
              onSubmitted: (_) => _submit(),
              decoration: const InputDecoration(
                labelText: 'Write in your own...',
                hintText: 'e.g., Johnny’s Wings',
                border: OutlineInputBorder(),
              ),
            ),
            const Spacer(),
            SizedBox(
              width: double.infinity,
              child: FilledButton.icon(
                icon: const Icon(Icons.check_rounded),
                onPressed: _submit,
                style: FilledButton.styleFrom(
                  padding: const EdgeInsets.symmetric(vertical: 14),
                  backgroundColor: const Color(0xFFF25C54),
                ),
                label: const Text(
                  'Done',
                  style: TextStyle(fontWeight: FontWeight.w700),
                ),
              ),
            ),
          ],
        ),
      ),
    );
  }
}

class _Cat {
  final String emoji;
  final String label;
  const _Cat(this.emoji, this.label);
}
